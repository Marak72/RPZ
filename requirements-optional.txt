# Необязательные зависимости.
#
# pypdf нужен ТОЛЬКО для автоматического распознавания индикаторов из PDF.
# Без него приложение полностью работоспособно: письма в PDF сохраняются,
# открываются в браузере, индикаторы к ним можно добавить вручную.
#
# Установка при доступном PyPI:
#     pip install -r requirements-optional.txt
#
# Если PyPI закрыт, но доступен GitHub:
#     pip install "pypdf @ git+https://github.com/py-pdf/pypdf@4.3.1"
#
# Либо из репозитория дистрибутива (ALT Linux):
#     apt-get install python3-module-pypdf
# и пересоздать окружение с доступом к системным пакетам:
#     python3 -m venv --system-site-packages venv

pypdf==4.3.1

# ldap3 и pywinrm нужны сервису «Узлы сети»:
#   ldap3   — чтение Active Directory (кто эта машина, в каком подразделении);
#   pywinrm — чтение аренд Windows DHCP (какой адрес какой машине выдан).
#
# Без них портал поднимается и работает, но раздел «Узлы сети» на любой
# запрос честно ответит, что библиотека не установлена. Обе — чистый Python,
# компилятор не нужен.
#
#     pip install ldap3 pywinrm
ldap3==2.9.1
pywinrm==0.5.0
